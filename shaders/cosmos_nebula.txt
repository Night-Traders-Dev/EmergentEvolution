#version 450

layout(local_size_x = 4, local_size_y = 4, local_size_z = 4) in;

struct NebulaComputeInput {
    vec4 seed_class_temp;
    vec4 atmosphere;
    vec4 activity;
    vec4 composition;
};

layout(std430, set = 0, binding = 0) readonly buffer InputBuffer {
    NebulaComputeInput entries[];
};

layout(rgba16f, set = 0, binding = 1) uniform readonly image3D prevField;
layout(rgba16f, set = 0, binding = 2) uniform writeonly image3D outField;

layout(push_constant) uniform PushConstants {
    uint body_count;
    float sim_time;
    float dt_hint;
    uint volume_dim;
} pc;

const int RENDER_NEBULA = 6;

vec3 hash33(vec3 p) {
    p = vec3(dot(p, vec3(127.1, 311.7, 74.7)),
             dot(p, vec3(269.5, 183.3, 246.1)),
             dot(p, vec3(113.5, 271.9, 124.6)));
    return -1.0 + 2.0 * fract(sin(p) * 43758.5453123);
}

float perlin3(vec3 p) {
    vec3 i = floor(p);
    vec3 f = fract(p);
    vec3 u = f * f * f * (f * (f * 6.0 - 15.0) + 10.0);

    float n000 = dot(hash33(i + vec3(0.0, 0.0, 0.0)), f - vec3(0.0, 0.0, 0.0));
    float n100 = dot(hash33(i + vec3(1.0, 0.0, 0.0)), f - vec3(1.0, 0.0, 0.0));
    float n010 = dot(hash33(i + vec3(0.0, 1.0, 0.0)), f - vec3(0.0, 1.0, 0.0));
    float n110 = dot(hash33(i + vec3(1.0, 1.0, 0.0)), f - vec3(1.0, 1.0, 0.0));
    float n001 = dot(hash33(i + vec3(0.0, 0.0, 1.0)), f - vec3(0.0, 0.0, 1.0));
    float n101 = dot(hash33(i + vec3(1.0, 0.0, 1.0)), f - vec3(1.0, 0.0, 1.0));
    float n011 = dot(hash33(i + vec3(0.0, 1.0, 1.0)), f - vec3(0.0, 1.0, 1.0));
    float n111 = dot(hash33(i + vec3(1.0, 1.0, 1.0)), f - vec3(1.0, 1.0, 1.0));

    float nx00 = mix(n000, n100, u.x);
    float nx10 = mix(n010, n110, u.x);
    float nx01 = mix(n001, n101, u.x);
    float nx11 = mix(n011, n111, u.x);
    float nxy0 = mix(nx00, nx10, u.y);
    float nxy1 = mix(nx01, nx11, u.y);
    return mix(nxy0, nxy1, u.z);
}

float fbm3(vec3 p) {
    float sum = 0.0;
    float amp = 0.55;
    float freq = 1.0;
    for (int i = 0; i < 4; ++i) {
        sum += amp * perlin3(p * freq);
        freq *= 2.02;
        amp *= 0.5;
    }
    return sum * 0.5 + 0.5;
}

float smooth_min(float a, float b, float k) {
    float h = clamp(0.5 + 0.5 * (b - a) / max(k, 1.0e-4), 0.0, 1.0);
    return mix(b, a, h) - k * h * (1.0 - h);
}

ivec3 clamp_coord(ivec3 c, int dim) {
    return clamp(c, ivec3(0), ivec3(dim - 1));
}

vec4 load_prev(ivec3 c, int dim) {
    return imageLoad(prevField, clamp_coord(c, dim));
}

float sample_prev_density(vec3 uvw, int dim) {
    ivec3 c = ivec3(uvw * float(dim));
    c = clamp_coord(c, dim);
    return load_prev(c, dim).x;
}

void aggregate_nebula_params(out float seed, out float dust, out float pressure,
                             out float haze, out float flow, out float cloud_detail, out float collapse) {
    seed = 0.0;
    dust = 0.0;
    pressure = 0.0;
    haze = 0.0;
    flow = 0.0;
    cloud_detail = 0.0;
    collapse = 0.0;
    float wsum = 0.0;
    uint limit = min(pc.body_count, 16u);
    for (uint i = 0u; i < limit; ++i) {
        NebulaComputeInput e = entries[i];
        if (int(e.seed_class_temp.y + 0.5) != RENDER_NEBULA) continue;
        float w = 1.0 + max(e.atmosphere.y, 0.0) * 0.01 + max(e.activity.z, 0.0) * 0.5;
        seed += e.seed_class_temp.x * w;
        dust += clamp(e.composition.w, 0.0, 1.0) * w;
        pressure += max(e.atmosphere.y, 0.0) * w;
        haze += clamp(e.atmosphere.z, 0.0, 2.5) * w;
        flow += clamp(e.activity.x, 0.0, 2.5) * w;
        cloud_detail += clamp(e.activity.y, 0.0, 2.5) * w;
        collapse += clamp(e.activity.z, 0.0, 2.5) * w;
        wsum += w;
    }
    if (wsum < 1.0e-5) {
        seed = 1.0;
        dust = 0.35;
        pressure = 30.0;
        haze = 0.8;
        flow = 0.7;
        cloud_detail = 0.8;
        collapse = 0.2;
        return;
    }
    float inv = 1.0 / wsum;
    seed *= inv;
    dust *= inv;
    pressure *= inv;
    haze *= inv;
    flow *= inv;
    cloud_detail *= inv;
    collapse *= inv;
}

float layered_density(vec3 p, float seed, float t, vec3 v_a, vec3 v_b, float collapse) {
    vec3 p_a = p + v_a * t + vec3(seed * 0.013, seed * 0.009, seed * 0.017);
    vec3 p_b = p + v_b * t + vec3(seed * 0.021, seed * 0.015, seed * 0.011);
    float d_a = fbm3(p_a * 2.2);
    float d_b = fbm3(p_b * 3.6);
    float additive = clamp(d_a * 0.72 + d_b * 0.58, 0.0, 1.0);
    float condense = 1.0 - smooth_min(1.0 - d_a, 1.0 - d_b, 0.32);
    float exponent = mix(1.15, 3.9, clamp(collapse * 0.6, 0.0, 1.0));
    return pow(clamp(additive * 0.62 + condense * 0.55, 0.0, 1.0), exponent);
}

vec3 base_velocity(vec3 p, float seed, float flow_strength, float t, out vec3 v_a, out vec3 v_b) {
    v_a = vec3(0.10, -0.06, 0.08) * (0.5 + flow_strength);
    v_b = vec3(-0.08, 0.09, -0.05) * (0.4 + flow_strength * 0.8);
    vec3 q = p + vec3(seed * 0.013, seed * 0.007, seed * 0.017);
    vec3 qa = q + v_a * t;
    vec3 qb = q + v_b * t;
    vec3 la = vec3(
        perlin3(qa * 1.9 + vec3(2.1, 0.8, 1.7)),
        perlin3(qa * 2.1 + vec3(0.4, 3.3, 2.6)),
        perlin3(qa * 1.7 + vec3(1.3, 2.7, 4.1)));
    vec3 lb = vec3(
        perlin3(qb * 2.4 + vec3(4.5, 1.7, 0.8)),
        perlin3(qb * 2.0 + vec3(0.9, 5.2, 1.4)),
        perlin3(qb * 2.8 + vec3(2.3, 0.5, 6.1)));
    return (la * 0.64 + lb * 0.48) * (0.14 + flow_strength * 0.62);
}

void main() {
    ivec3 gid = ivec3(gl_GlobalInvocationID.xyz);
    int dim = int(pc.volume_dim);
    if (gid.x >= dim || gid.y >= dim || gid.z >= dim) return;

    float seed;
    float dust;
    float pressure;
    float haze;
    float flow;
    float cloud_detail;
    float collapse;
    aggregate_nebula_params(seed, dust, pressure, haze, flow, cloud_detail, collapse);

    vec3 uvw = (vec3(gid) + vec3(0.5)) / float(dim);
    vec3 p = uvw * 2.0 - 1.0;

    vec4 prev = load_prev(gid, dim);
    float prev_density = prev.x;
    vec3 prev_vel = prev.yzw;

    vec3 va;
    vec3 vb;
    vec3 vel_base = base_velocity(p, seed, flow + cloud_detail * 0.25, pc.sim_time, va, vb);

    float dt = max(pc.dt_hint, 1.0 / 240.0);
    vec3 pos_prev = p - prev_vel * dt * 0.35; // semi-Lagrangian advection
    float advected_density = sample_prev_density(pos_prev * 0.5 + 0.5, dim);

    // Vorticity from previous velocity field
    vec3 vx1 = load_prev(gid + ivec3(1,0,0), dim).yzw;
    vec3 vx0 = load_prev(gid - ivec3(1,0,0), dim).yzw;
    vec3 vy1 = load_prev(gid + ivec3(0,1,0), dim).yzw;
    vec3 vy0 = load_prev(gid - ivec3(0,1,0), dim).yzw;
    vec3 vz1 = load_prev(gid + ivec3(0,0,1), dim).yzw;
    vec3 vz0 = load_prev(gid - ivec3(0,0,1), dim).yzw;
    vec3 vort = vec3(
        (vy1.z - vy0.z) - (vz1.y - vz0.y),
        (vz1.x - vz0.x) - (vx1.z - vx0.z),
        (vx1.y - vx0.y) - (vy1.x - vy0.x)) * 0.5;

    // Attraction toward local density peaks
    float dx = load_prev(gid + ivec3(1,0,0), dim).x - load_prev(gid - ivec3(1,0,0), dim).x;
    float dy = load_prev(gid + ivec3(0,1,0), dim).x - load_prev(gid - ivec3(0,1,0), dim).x;
    float dz = load_prev(gid + ivec3(0,0,1), dim).x - load_prev(gid - ivec3(0,0,1), dim).x;
    vec3 grad = vec3(dx, dy, dz) * 0.5;
    vec3 attract = (dot(grad, grad) > 1.0e-6) ? normalize(grad) : vec3(0.0);

    vec3 vel = mix(prev_vel, vel_base, 0.18 + cloud_detail * 0.06);
    vel += vort * (0.015 + cloud_detail * 0.020);
    vel += attract * (0.010 + collapse * 0.035);

    float d_layer = layered_density(p, seed, pc.sim_time, va, vb, collapse);
    float density_target = d_layer *
        (0.55 + haze * 0.38 + pressure * 0.0012 + cloud_detail * 0.20) *
        (0.72 + dust * 0.42);

    float density = mix(advected_density, density_target, 0.14 + collapse * 0.08);
    density = mix(density, prev_density, 0.12);
    density = clamp(density, 0.0, 4.0);

    imageStore(outField, gid, vec4(density, vel));
}
